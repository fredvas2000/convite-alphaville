// envia a imagem comprimida pro Apps Script via POST (form-urlencoded, // pra não esbarrar em bloqueio de CORS) — separado do helper GET porque // o payload da imagem é grande demais pra caber numa URL function postarFotoNoBackend(nome, legenda, imagemBase64) { if (!EVENT.pontuacaoUrl) return Promise.resolve(null); const corpo = new URLSearchParams({ action: 'postarFoto', nome, legenda: legenda || '', imagem: imagemBase64 }); return fetch(EVENT.pontuacaoUrl, { method: 'POST', body: corpo }) .then(r => r.json()) .catch(() => null); } const fotoBtn = document.getElementById('fotoBtn'); const fotoInput = document.getElementById('fotoInput'); const fotoUploadTexto = document.getElementById('fotoUploadTexto'); if (fotoBtn && fotoInput) { fotoBtn.addEventListener('click', () => fotoInput.click()); fotoInput.addEventListener('change', () => { const arquivo = fotoInput.files[0]; if (!arquivo) return; const nomeConvidado = getNomeConvidado(); if (!nomeConvidado) { if (fotoUploadTexto) fotoUploadTexto.textContent = 'Confirme sua presença antes de enviar uma foto.'; fotoInput.value = ''; return; } fotoBtn.disabled = true; fotoBtn.textContent = 'enviando...'; comprimirImagem(arquivo, 1200, 0.75) .then(imagemComprimida => postarFotoNoBackend(nomeConvidado, '', imagemComprimida)) .then(resultado => { fotoInput.value = ''; fotoBtn.disabled = false; if (!resultado || resultado.erro) { fotoBtn.textContent = 'enviar uma foto'; if (fotoUploadTexto) fotoUploadTexto.textContent = 'Não deu pra enviar agora — tenta de novo?'; return; } fotoBtn.textContent = 'foto enviada ✓'; if (fotoUploadTexto) fotoUploadTexto.textContent = 'Entrou na parede! Pode mandar outra quando quiser.'; atualizarContadorEstalecas(resultado); atualizarGaleriaDoBackend(); setTimeout(() => { fotoBtn.textContent = 'enviar uma foto'; }, 2500); }); }); } // auto-inicia à noite se já passou das 18h no navegador do usuário (só um toque de contexto) const hour = new Date().getHours(); if(hour >= 18 || hour < 6){ body.classList.add('night'); label.textContent = 'noite'; }
